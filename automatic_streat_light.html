Arduino Traffic Light

An automatic street light system that uses an LDR sensor to turn the light ON in darkness and OFF in daylight.

Components

Arduino UNO Photoresistor 10 kΩ Resistor 330 Ω Resistor Green LED Jumper Wires

Tinkercad Simulation

Arduino Code

automatic_streat_light.ino 

int ldrPin = A0;
int ledPin = 10;

void setup() {
  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int lightValue = analogRead(ldrPin);

  Serial.println(lightValue);

  // Turn ON street light when it is dark
  if (lightValue < 500) {
    digitalWrite(ledPin, HIGH);
  } 
  else {
    digitalWrite(ledPin, LOW);
  }

  delay(100);
}
