Arduino Traffic Light

An automatic street light system that uses an LDR sensor to turn the light ON in darkness and OFF in daylight.

Components

Arduino UNO Temperature Sensor [TMP36] PCF8574-based, 32 (0x20) LCD 16 x 2 (I2C) TIP120 DC Motor 1 kΩ Resistor Jumper Wires

Tinkercad Simulation

Arduino Code

automatic_fan.ino 

#include 
#include 

LiquidCrystal_I2C lcd(0x20, 16, 2);

const int tempPin = A0;
const int motorPin = 9;

void setup() {
  lcd.init();
  lcd.backlight();

  pinMode(motorPin, OUTPUT);

  lcd.setCursor(0, 0);
  lcd.print("TEMP FAN");
  delay(1000);
}

void loop() {

  // Read TMP36
  int sensorValue = analogRead(tempPin);

  float voltage = sensorValue * 5.0 / 1023.0;

  // TMP36
  float temperature = (voltage - 0.5) * 100.0;

  // Convert temperature to motor speed
  int rpm;

  if (temperature < 20) {
    rpm = 0;
  }
  else if (temperature >= 20 && temperature < 30) {
    rpm = 1000;
  }
  else if (temperature >= 30 && temperature < 40) {
    rpm = 2000;
  }
  else if (temperature >= 40 && temperature < 50) {
    rpm = 3000;
  }
  else {
    rpm = 4000;
  }

  // Convert RPM to PWM
  int pwm = map(rpm, 0, 4000, 0, 255);

  analogWrite(motorPin, pwm);

  // LCD
  lcd.setCursor(0, 0);
  lcd.print("Temp: ");
  lcd.print(temperature, 1);
  lcd.write(223);
  lcd.print("C   ");

  lcd.setCursor(0, 1);
  lcd.print("RPM: ");
  lcd.print(rpm);
  lcd.print("    ");

  delay(500);
}
