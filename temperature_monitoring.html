Arduino Traffic Light

An automatic street light system that uses an LDR sensor to turn the light ON in darkness and OFF in daylight.

Components

Arduino UNO Temperature Sensor [TMP36] PCF8574-based, 32 (0x20) LCD 16 x 2 (I2C) Jumper Wires

Tinkercad Simulation

Arduino Code

temperature_monitoring.ino 

#include 
#include 

LiquidCrystal_I2C lcd(0x20, 16, 2);

const int tempPin = A0;

void setup() {
  Serial.begin(9600);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("TEMP MONITOR");

  delay(1000);
}

void loop() {
  int reading = analogRead(tempPin);

  float voltage = reading * 5.0 / 1023.0;
  float temperature = (voltage - 0.5) * 100.0;

  lcd.setCursor(0, 1);
  lcd.print("Temp: ");
  lcd.print(temperature, 1);
  lcd.write(223);
  lcd.print("C   ");

  Serial.print("Temperature: ");
  Serial.print(temperature, 1);
  Serial.println(" C");

  delay(1000);
}
